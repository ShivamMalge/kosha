Updated `cli.py`. Precedence chain is flags → env → TOML file → built-in defaults, applied by layering dicts in reverse-priority order.

```python
"""Command-line entry point with layered configuration.

Precedence, highest first: command-line flags, environment variables,
TOML config file, built-in defaults.
"""

import argparse
import os
import tomllib
from pathlib import Path

DEFAULTS = {"host": "127.0.0.1", "port": 8000}
DEFAULT_CONFIG_PATH = Path("config.toml")


def build_parser():
    parser = argparse.ArgumentParser(prog="server", description="Run the server.")
    parser.add_argument(
        "--config",
        type=Path,
        help="TOML config file (env: APP_CONFIG, default: ./config.toml if present)",
    )
    parser.add_argument(
        "--host",
        help="Interface to bind to (env: APP_HOST, default: 127.0.0.1)",
    )
    parser.add_argument(
        "--port",
        type=int,
        help="Port to listen on (env: APP_PORT, default: 8000)",
    )
    return parser


def _port(value, source, parser):
    try:
        return int(value)
    except (TypeError, ValueError):
        parser.error(f"invalid port from {source}: {value!r}")


def _from_file(path, parser):
    """Read host/port from a TOML file. Returns only the keys it actually finds."""
    explicit = path is not None or "APP_CONFIG" in os.environ
    path = path or Path(os.environ.get("APP_CONFIG", DEFAULT_CONFIG_PATH))

    try:
        with open(path, "rb") as handle:
            table = tomllib.load(handle)
    except FileNotFoundError:
        if explicit:
            parser.error(f"config file not found: {path}")
        return {}  # the implicit default is allowed to be absent
    except tomllib.TOMLDecodeError as exc:
        parser.error(f"{path}: {exc}")

    unknown = set(table) - set(DEFAULTS)
    if unknown:
        parser.error(f"{path}: unknown key(s): {', '.join(sorted(unknown))}")

    values = {k: v for k, v in table.items() if k in DEFAULTS}
    if "port" in values:
        values["port"] = _port(values["port"], path, parser)
    return values


def _from_env(parser):
    values = {}
    if "APP_HOST" in os.environ:
        values["host"] = os.environ["APP_HOST"]
    if "APP_PORT" in os.environ:
        values["port"] = _port(os.environ["APP_PORT"], "APP_PORT", parser)
    return values


def resolve(argv=None):
    parser = build_parser()
    args = parser.parse_args(argv)

    config = dict(DEFAULTS)
    config.update(_from_file(args.config, parser))
    config.update(_from_env(parser))
    config.update({k: v for k in DEFAULTS if (v := getattr(args, k)) is not None})
    return argparse.Namespace(**config)


def main(argv=None):
    args = resolve(argv)
    print(f"Listening on {args.host}:{args.port}")


if __name__ == "__main__":
    main()
```

Config file shape is flat:

```toml
host = "0.0.0.0"
port = 8080
```

Three things worth knowing:

- The flags now default to `None` rather than to real values — that's what makes "flag was not passed" distinguishable from "flag was passed the default value." The visible defaults moved into the help strings, so `--help` still shows them.
- `--config` is itself layered (`--config` > `APP_CONFIG` > `./config.toml`), but only two of those three are errors when missing. An explicitly named file that doesn't exist is a mistake; the implicit `./config.toml` is optional by design.
- `tomllib` is stdlib on Python 3.11+. On older versions, `pip install tomli` and swap to `import tomli as tomllib`.
